;; Ground cover (M2-style grass) for RTW battle maps.
;;
;; This file drives the static-buffer grass system that replaced the old per-frame
;; descr_grass.txt renderer. Grass is placed procedurally from the battle map's ground
;; texture types at load; nothing here touches map or save formats.
;;
;; The atlas pages are shared with M2 (data/new_vegetation/grass/...). Climate blocks
;; use RTW climate names (see descr_climates.txt); atlas keys are the battlefield
;; geography ground types (identical between RTW and M2 - this is NOT the old
;; descr_grass.txt ground-type list). Empty or missing blocks get no ground cover.
;;

ground_cover
{
	; rendering settings
	;
	; fade_out
	; {
	;	[low, medium or high]
	;	{
	;		lowres_begin	[low res begin distance - fades off after this]
	;		lowres_end		[low res end distance - not rendered after this]
	;		highres_begin	[high res begin distance - fades off after this]
	;		highres_end		[high res end distance - not rendered after this]
	;	}
	; }
	;
	; texel_resolution	[base size of 1 texel in metres]
	; highres_density	[density scale factor for high res ground cover]
	; highres_height_scale	[scale applied to highres vege height]
	;
	rendering
	{
		fade_out
		{

				lowres_end		130

		highres_density		5		; maximum is 7, minimum is 1
